   kubectl get pods -n {{ .Release.Namespace }} -l app.kubernetes.io/name={{ include "training-operator.name" . }}

   kubectl get svc -n {{ .Release.Namespace }} -l app.kubernetes.io/name={{ include "training-operator.name" . }}

   kubectl get crds | grep kubeflow.org

   kubectl get validatingwebhookconfigurations -l app.kubernetes.io/name={{ include "training-operator.name" . }}
